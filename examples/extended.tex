\section{Algorithm}

\begin{algorithm}
\caption{Euclidean Algorithm}
\begin{algorithmic}
\STATE $r \gets a \bmod b$
\WHILE{$r \neq 0$}
\STATE $a \gets b$
\STATE $b \gets r$
\STATE $r \gets a \bmod b$
\ENDWHILE
\RETURN $b$
\end{algorithmic}
\end{algorithm}
